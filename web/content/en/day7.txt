import re

HEAD = re.compile(r"^(\d+(?:\.\d+)*)\s+([A-Z].*)$")

def split_clauses(text):
    clauses = []
    current = None
    for line in text.split("\n"):
        m = HEAD.match(line.strip())
        if m:
            current = {"id": m.group(1), "heading": m.group(2), "lines": []}
            clauses.append(current)
        elif current is not None:
            current["lines"].append(line)
    return [{"id": c["id"], "heading": c["heading"], "text": "\n".join(c["lines"]).strip()} for c in clauses]

def review_all(clauses):
    results = []
    for c in clauses:
        v = classify(f"Review clause {c['id']}: {c['text']}")
        results.append({"id": c["id"], "verdict": v["verdict"], "risk": v["risk_level"]})

def estimate_cost(usage, price):
    cost = (
        usage["prompt_cache_miss_tokens"] * price["input"]
        + usage["prompt_cache_hit_tokens"] * price["cache"]
        + usage["completion_tokens"] * price["output"]
    ) / 1_000_000
    return round(cost, 6)

type Clause = { id: string; heading: string; text: string };
const HEAD = /^(\d+(?:\.\d+)*)\s+([A-Z].*)$/;

function splitClauses(text: string): Clause[] {
  const found: { id: string; heading: string; lines: string[] }[] = [];
  for (const line of text.split("\n")) {
    const m = HEAD.exec(line.trim());
    if (m) {
      found.push({ id: m[1], heading: m[2], lines: [] });
    } else if (found.length > 0) {
      found[found.length - 1].lines.push(line);
    }
  }
  return found.map((c) => ({ id: c.id, heading: c.heading, text: c.lines.join("\n").trim() }));
}

  const res = await fetch(API_URL, { method: "POST", headers: HEADERS, body: JSON.stringify({ model: "deepseek-flash", messages: [{ role: "user", content: text }], response_format: { type: "json_object" }, thinking: { type: "disabled" } }) });

async function reviewAll(clauses: Clause[]): Promise<{ counts: Record<string, number>; needs_attention: string[] }> {
  const results: { id: string; verdict: string; risk: string }[] = [];
  for (const c of clauses) {
    const v = await classify(`Review clause ${c.id}: ${c.text}`);
    results.push({ id: c.id, verdict: v.verdict, risk: v.risk_level });
  }
  const counts: Record<string, number> = {};
  for (const r of results) {
    counts[r.verdict] = (counts[r.verdict] ?? 0) + 1;

const API_URL = "https://api.deepseek.com/v1/chat/completions";
const HEADERS = { Authorization: `Bearer ${process.env.DEEPSEEK_API_KEY}` };
const TOOLS = [{ type: "function", function: { name: "lookup_playbook", description: "Return the customer's rule for a topic", parameters: { type: "object", properties: { topic: { type: "string" } }, required: ["topic"] } } }];
const IMPL: Record<string, (args: any) => string> = {
  lookup_playbook: (args) => `rule for ${args.topic}: cap at 100% of annual fees`,
};

async function miniAgent(question: string, maxSteps = 3): Promise<string> {
  const messages: any[] = [{ role: "user", content: question }];
  for (let step = 0; step < maxSteps; step++) {
    const res = await fetch(API_URL, { method: "POST", headers: HEADERS, body: JSON.stringify({ model: "deepseek-flash", messages, tools: TOOLS, thinking: { type: "disabled" } }) });
    const choice = (await res.json()).choices[0];
    if (choice.finish_reason !== "tool_calls") {
      return choice.message.content;
    }
    const call = choice.message.tool_calls[0];
    const result = IMPL[call.function.name](JSON.parse(call.function.arguments));
    messages.push(choice.message);
    messages.push({ role: "tool", tool_call_id: call.id, content: result });
  }
  return "STOPPED";
}

def parse_and_count(raw_list):
    valid = 0

import os
import requests

URL = "https://api.deepseek.com/v1/chat/completions"

def classify_one(text):
    headers = {"Authorization": f"Bearer {os.getenv('DEEPSEEK_API_KEY')}"}

const URL_ = "https://api.deepseek.com/v1/chat/completions";

async function classifyOne(text: string): Promise<string> {
  const body = JSON.stringify({ model: "deepseek-flash", messages: [{ role: "user", content: text }], response_format: { type: "json_object" }, thinking: { type: "disabled" } });
  for (let i = 0; i < 3; i++) {
    const res = await fetch(URL_, { method: "POST", headers: { Authorization: `Bearer ${process.env.DEEPSEEK_API_KEY}` }, body });
    if (res.status === 200) {
      const data = await res.json();
      return JSON.parse(data.choices[0].message.content).verdict;
    }
    if (res.status !== 429) break;
  }
  throw new Error("classify failed");
}

async function classifyAll(texts: string[]): Promise<string[]> {
  const out: string[] = [];
  for (const t of texts) {
    out.push(await classifyOne(t));
  }
  return out;
}
